let t = fs.readFileSync(file, "utf8");
const crlf = t.includes("\r\n");
const N = (s) => (crlf ? s.replace(/\r?\n/g, "\r\n") : s);
function rep(a, b, count = 1) {
  a = N(a);
  b = N(b);
  const n = t.split(a).length - 1;
  if (n !== count) throw new Error(`found ${n}x (want ${count}): ${a.slice(0, 90)}`);
  t = t.split(a).join(b);
}

rep(
  `    /// The expected registry, PSO by PSO. A hand-written twin of`,
  `    /// THE NEAR SUN CASCADES' SWITCH (2026-09-27, docs/design/sun-cascades.md).
    /// The cascade function is called in exactly three places before
    /// fs_shadow, each the line right after its guard
    /// \`if (HAS_SUN_CASCADES_BRANCH && shadow_u.near.ctl.x > 0.5) {\`, which is
    /// the first statement after frag_prologue in fs_surface, fs_terrain and
    /// fs_vegetation; frag_tail and frag_prologue read no lighting switch; and
    /// the switch is live in exactly those three classes' PSOs and dead in
    /// every other row (water, shell, cloud, the three sun-shadow PSOs). Seen
    /// fail first by calling the function unguarded from frag_tail (the use
    /// count named the extra site), then restored.
    #[test]
    fn the_sun_cascade_sample_is_guarded_at_three_sites_only() {
        let main = code_only(class_entries_side());
        let guard = format!("if ({LIGHTING_SUN_CASCADES} && shadow_u.near.ctl.x > 0.5) {{");
        let sites: Vec<usize> = main.match_indices(&guard).map(|(p, _)| p).collect();
        let entries = ["fs_surface", "fs_terrain", "fs_vegetation"];
        assert_eq!(sites.len(), entries.len(), "one {guard:?} per lit opaque class entry, found {}", sites.len());
        for entry in entries {
            let (start, end) = function_body(&main, entry);
            let here: Vec<usize> = sites.iter().copied().filter(|&p| p > start && p < end).collect();
            assert_eq!(here.len(), 1, "{entry} must hold exactly one sun-cascade guard");
            // The guard comes right after the prologue: nothing but the
            // prologue call between the entry's opening brace and it.
            let before = main[start..here[0]].trim();
            assert!(
                before.starts_with("var s = frag_prologue(in);") && before.matches(';').count() == 1,
                "{entry}: the sun-cascade guard must be the first statement after frag_prologue, found {before:?}"
            );
            let next_line = main[here[0] + guard.len()..].lines().nth(1).unwrap_or("").trim();
            assert!(
                next_line.starts_with("s.sun_vis = sun_cascade_visibility("),
                "{entry}: the line after the guard must set s.sun_vis from sun_cascade_visibility, found {next_line:?}"
            );
        }
        let uses: Vec<usize> = identifier_sites(&main, "sun_cascade_visibility")
            .into_iter()
            .filter(|&p| !main[..p].ends_with("fn "))
            .collect();
        assert_eq!(uses.len(), 3, "sun_cascade_visibility must be called at exactly the three guarded sites");
        for helper in ["frag_prologue", "frag_tail"] {
            let read = switches_read_by(&main, helper);
            assert!(!read.contains(&LIGHTING_SUN_CASCADES.to_string()), "{helper} must not read {LIGHTING_SUN_CASCADES}");
        }
        let owners: Vec<ShaderClass> =
            ShaderClass::ALL.iter().copied().filter(|c| c.live_lighting().contains(&LIGHTING_SUN_CASCADES)).collect();
        assert_eq!(owners, vec![ShaderClass::Surface, ShaderClass::Terrain, ShaderClass::Vegetation]);
        for row in PSO_REGISTRY {
            let live = row.class.is_some_and(|c| owners.contains(&c));
            assert_eq!(!row.dead.contains(&LIGHTING_SUN_CASCADES), live, "{}: sun cascades live = {live}", row.label);
        }
    }

    /// The expected registry, PSO by PSO. A hand-written twin of`
